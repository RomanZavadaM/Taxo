name: Build Taxo v8.65 candidate Windows

on:
  workflow_dispatch:
  push:
    branches:
      - candidate-v8.65
    paths:
      - 'main.py'
      - 'tachograph.py'
      - 'attestation_render.py'
      - 'Бланк підтвердження.docx'
      - 'attestation_visual_template.pdf'
      - 'requirements.txt'
      - 'Taxo.spec'
      - 'installer/**'
      - 'BUILD_WINDOWS.bat'
      - 'BUILD_INSTALLER.bat'
      - '.github/workflows/build-windows-v8.65-candidate.yml'
      - 'VERSION.txt'

permissions:
  contents: read

jobs:
  build-windows:
    runs-on: windows-latest
    env:
      PYTHONUTF8: '1'
      MAIN_SHA256: 8373d80611f2bcb3d5efb6c29f17bb9363dd8436dc0ccf010fa4941063e8d5f7
      TACHO_SHA256: 83c916457c6602e9ed93a55fdc0578e91775e073dad12da94ccd31d1e81fa108
      RENDER_SHA256: b65178e0bf18207e75e10ebfb6c2896397e685184228ece70efb7615d4215c88
      TEMPLATE_SHA256: 47aeb545509af8d43102f76de772838b36de1fa85a2d09bae2d2ce7798358ec0
      VISUAL_TEMPLATE_SHA256: 6e5d79bebdeed68c79e49d438d37dbd165e2316fda3233b3442911a9a3b61151

    steps:
      - name: Checkout candidate
        uses: actions/checkout@v4

      - name: Set up Python 3.13 x64
        uses: actions/setup-python@v5
        with:
          python-version: '3.13'
          architecture: 'x64'

      - name: Verify candidate source hashes
        shell: pwsh
        run: |
          $checks = @{
            'main.py' = $env:MAIN_SHA256
            'tachograph.py' = $env:TACHO_SHA256
            'attestation_render.py' = $env:RENDER_SHA256
            'Бланк підтвердження.docx' = $env:TEMPLATE_SHA256
            'attestation_visual_template.pdf' = $env:VISUAL_TEMPLATE_SHA256
          }
          foreach ($file in $checks.Keys) {
            if (!(Test-Path $file)) { throw "Missing required file: $file" }
            $actual = (Get-FileHash $file -Algorithm SHA256).Hash.ToLowerInvariant()
            if ($actual -ne $checks[$file]) { throw "SHA-256 mismatch for $file : $actual" }
            Write-Host "OK $file $actual"
          }

      - name: Compile source
        shell: pwsh
        run: python -m py_compile main.py tachograph.py attestation_render.py

      - name: Install Python dependencies
        shell: pwsh
        run: |
          python -m pip install --upgrade pip
          python -m pip install -r requirements.txt
          python -m pip install --upgrade pyinstaller

      - name: Smoke import renderer
        shell: pwsh
        run: python -c "import attestation_render, fitz, reportlab; print('renderer imports OK')"

      - name: Build portable application
        shell: pwsh
        run: python -m PyInstaller --noconfirm --clean Taxo.spec

      - name: Verify portable bundle
        shell: pwsh
        run: |
          if (!(Test-Path 'dist/Taxo/Taxo.exe')) { throw 'Taxo.exe not found' }
          if (!(Test-Path 'dist/Taxo/_internal/Бланк підтвердження.docx')) { throw 'DOCX template not found in bundle' }
          if (!(Test-Path 'dist/Taxo/_internal/attestation_visual_template.pdf')) { throw 'Visual PDF template not found in bundle' }
          $db = Get-ChildItem 'dist/Taxo' -Recurse -File -Include *.db,*.sqlite,*.sqlite3
          if ($db) { throw "Database unexpectedly bundled: $($db.FullName -join ', ')" }

      - name: Create portable ZIP
        shell: pwsh
        run: Compress-Archive -Path 'dist/Taxo' -DestinationPath 'Taxo_v8_65_Windows_x64_Portable.zip' -Force

      - name: Install Inno Setup
        shell: pwsh
        run: choco install innosetup --no-progress -y

      - name: Build installer
        shell: pwsh
        run: |
          New-Item -ItemType Directory -Force -Path release_out | Out-Null
          $candidates = @(
            (Join-Path ${env:ProgramFiles(x86)} 'Inno Setup 6\ISCC.exe'),
            (Join-Path $env:ProgramFiles 'Inno Setup 6\ISCC.exe'),
            (Join-Path ${env:ProgramFiles(x86)} 'Inno Setup 7\ISCC.exe'),
            (Join-Path $env:ProgramFiles 'Inno Setup 7\ISCC.exe')
          )
          $iscc = $candidates | Where-Object { Test-Path $_ } | Select-Object -First 1
          if (!$iscc) { throw 'ISCC.exe not found' }
          & $iscc 'installer\Taxo.iss'
          if ($LASTEXITCODE -ne 0) { throw "Inno Setup failed: $LASTEXITCODE" }
          if (!(Test-Path 'release_out/Taxo_v8_65_Setup_Windows_x64.exe')) { throw 'Installer not found' }

      - name: Create checksums
        shell: pwsh
        run: |
          $files = @(
            'release_out/Taxo_v8_65_Setup_Windows_x64.exe',
            'Taxo_v8_65_Windows_x64_Portable.zip'
          )
          $lines = foreach ($f in $files) {
            $h = (Get-FileHash $f -Algorithm SHA256).Hash.ToLowerInvariant()
            "$h  $([IO.Path]::GetFileName($f))"
          }
          $lines | Set-Content -Encoding ascii 'SHA256SUMS_WINDOWS.txt'
          Get-Content 'SHA256SUMS_WINDOWS.txt'

      - name: Upload candidate Windows artifacts
        uses: actions/upload-artifact@v4
        with:
          name: Taxo-v8.65-Windows-candidate
          path: |
            release_out/Taxo_v8_65_Setup_Windows_x64.exe
            Taxo_v8_65_Windows_x64_Portable.zip
            SHA256SUMS_WINDOWS.txt
          if-no-files-found: error
          retention-days: 30
